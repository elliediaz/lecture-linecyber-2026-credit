
```python
# 셀 2: KAMP CNC 공구마모 데이터 로드
df = pd.read_csv('data/part2_cnc_tool_wear.csv')

# 특성과 타겟 분리
feature_cols = ['x_load', 'y_load', 'z_load', 'spindle_speed', 'feed_rate']
X = df[feature_cols]
y = df['tool_wear_class']

class_names = ['낮음', '중간', '높음']

print("=== KAMP CNC 공구마모 데이터셋 정보 ===")
print(f"샘플 수: {len(X)}")
```

```python
# 셀 3: 학습/테스트 분할
X_train, X_test, y_train, y_test = train_test_split(

)
print(f"학습 데이터: {len(X_train)}개")
print(f"테스트 데이터: {len(X_test)}개")
```

```python
# 셀 4: 의사결정나무 모델 학습

print("모델 학습 완료!")
```

```python
# 셀 5: 정확도 확인
train_acc = model.score(X_train, y_train)

print(f"테스트 정확도: {test_acc:.2%}")
```


```python
# 셀 1: 라이브러리 불러오기
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

# TensorFlow 경고 숨기기
import os
os.environ['TF_CPP_MIN_LOG_LEVEL'] = '2'

import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Dropout

print(f"TensorFlow 버전: {tf.__version__}")
print("라이브러리 불러오기 완료!")
```

```python
# 셀 2: KAMP 프레스기 품질 데이터 로드
df = pd.read_csv('data/part3_press_quality.csv')

# 특성과 타겟 분리
feature_cols = [col for col in df.columns if col != 'quality']
X = df[feature_cols].values
y = df['quality'].values

# 정규화
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 분할
X_train, X_test, y_train, y_test = train_test_split(
    X_scaled, y, test_size=0.2, random_state=42
)

print("=== KAMP 프레스기 품질 데이터 정보 ===")
print(f"특성: {feature_cols}")
print(f"특성 수: {len(feature_cols)}")
print(f"학습 데이터: {len(X_train)}개")
print(f"테스트 데이터: {len(X_test)}개")
print(f"불량률: {y.mean():.2%}")
```

```python
# 셀 3: MLP 모델 생성
model = Sequential([
    Dense(64, activation='relu', input_shape=(X_train.shape[1],)),
    Dropout(0.3),
    Dense(32, activation='relu'),
    Dropout(0.2),
    Dense(1, activation='sigmoid')
])

model.compile(
    optimizer='adam',
    loss='binary_crossentropy',
    metrics=['accuracy']
)

print("모델 구조:")
model.summary()
```

```python
# 셀 4: 모델 학습
history = model.fit(
    X_train, y_train,
    epochs=50,
    batch_size=32,
    validation_split=0.2,
    verbose=1
)
print("\n학습 완료!")
```

```python
# 셀 5: 학습 곡선 시각화 및 저장
plt.figure(figsize=(10, 4))

# 손실 곡선

plt.plot(history.history['val_loss'], label='Val Loss')

plt.ylabel('Loss')
plt.legend()

plt.title('Accuracy Curve')
plt.xlabel('Epoch')
plt.ylabel('Accuracy')
plt.legend()

plt.tight_layout()
plt.savefig('learning_curve.png', dpi=150)
plt.show()
print("학습 곡선 저장 완료: learning_curve.png")
```

```python
# 셀 6: 최종 성능 확인
loss, accuracy = model.evaluate(X_test, y_test, verbose=0)
print(f"\n=== 최종 성능 ===")
print(f"테스트 손실: {loss:.4f}")
print(f"테스트 정확도: {accuracy:.2%}")
```
